// ========== 搜索模块 ==========
import { Article } from '../../models/article-model.js';
import { UIArticles } from './articles.js';
import { UIDirectory } from './directory.js';
import { UIHelpers } from './helpers.js';
 
export const UISearch = {
  searchInput: null,
  searchResults: [],
  searchCurrentIndex: -1,
  searchKeyword: '',
  directoryTreeContainer: null,
 
  init: function (searchInputEl, treeContainer) {
    console.log('[UISearch] 初始化...');
    this.searchInput = searchInputEl;
    this.directoryTreeContainer = treeContainer;
    this.bindEvents();
    console.log('[UISearch] 初始化完成');
  },
 
  bindEvents: function () {
    if (!this.searchInput) {
      console.warn('[UISearch] 搜索框元素不存在');
      return;
    }
 
    this.searchInput.disabled = false;
    this.searchInput.style.pointerEvents = 'auto';
    this.searchInput.style.opacity = '1';
    this.searchInput.style.zIndex = '100';
 
    this.searchInput.removeEventListener('focus', this._focusHandler);
    this.searchInput.removeEventListener('input', this._inputHandler);
    this.searchInput.removeEventListener('keydown', this._keydownHandler);
 
    this._focusHandler = () => {
      // 使用导入的 Sidebar（但需要从外部传入，这里通过全局 UI 引用处理，或直接用 import）
      // 为了解耦，使用事件或直接访问 DOM 状态。这里保留原逻辑，通过 window._UISidebar 兼容
      if (window._UISidebar && window._UISidebar.sidebarCollapsed) {
        window._UISidebar.toggleCollapse();
      }
    };
 
    this._inputHandler = (e) => {
      const keyword = e.target.value.trim();
      this.searchKeyword = keyword;
      this.performSearch(keyword);
    };
 
    this._keydownHandler = (e) => {
      if (e.key === 'Enter') {
        e.preventDefault();
        this.expandSearchResults();
        setTimeout(() => {
          if (this.searchInput) this.searchInput.focus();
        }, 100);
      }
      if (e.key === 'ArrowDown') {
        e.preventDefault();
        this.navigateSearchResult(1);
        setTimeout(() => {
          if (this.searchInput) this.searchInput.focus();
        }, 50);
      }
      if (e.key === 'ArrowUp') {
        e.preventDefault();
        this.navigateSearchResult(-1);
        setTimeout(() => {
          if (this.searchInput) this.searchInput.focus();
        }, 50);
      }
      if (e.key === 'Escape') {
        this.clearSearchHighlights();
        if (this.searchInput) this.searchInput.blur();
      }
    };
 
    this.searchInput.addEventListener('focus', this._focusHandler);
    this.searchInput.addEventListener('input', this._inputHandler);
    this.searchInput.addEventListener('keydown', this._keydownHandler);
    console.log('[UISearch] 事件绑定完成');
  },
 
  performSearch: function (keyword) {
    const visibleArticles = Article.getVisibleArticles();
    this.searchResults = [];
    this.searchCurrentIndex = -1;
    this.clearSearchHighlights();
 
    if (!keyword || keyword.length < 1) {
      UIDirectory.updateTree(visibleArticles);
      if (this.searchInput) {
        this.searchInput.placeholder = '🔍 搜索角色...';
      }
      return;
    }
 
    const searchLower = keyword.toLowerCase();
    const filtered = visibleArticles.filter(function (article) {
      return (
        (article.title && article.title.toLowerCase().includes(searchLower)) ||
        (article.content && article.content.toLowerCase().includes(searchLower))
      );
    });
 
    this.searchResults = filtered;
    UIDirectory.updateTree(filtered);
 
    if (filtered.length > 0) {
      if (this.searchInput) {
        this.searchInput.placeholder = '找到 ' + filtered.length + ' 个结果';
      }
    } else {
      if (this.searchInput) {
        this.searchInput.placeholder = '🔍 无匹配结果';
      }
    }
  },
 
  navigateSearchResult: function (direction) {
    if (this.searchResults.length === 0) return;
    this.clearSearchHighlights();
 
    this.searchCurrentIndex =
      (this.searchCurrentIndex + direction + this.searchResults.length) % this.searchResults.length;
    const article = this.searchResults[this.searchCurrentIndex];
 
    if (article && UIArticles) {
      const targetId = 'article-card-' + article.id;
      const targetElement = document.getElementById(targetId);
 
      if (!targetElement) {
        const allVisible = Article.getVisibleArticles();
        const index = allVisible.findIndex(function (a) {
          return a.id === article.id;
        });
        if (index !== -1) {
          const targetPage = Math.floor(index / UIArticles.pageSize) + 1;
          if (targetPage > UIArticles.currentPage) {
            UIArticles.currentPage = targetPage;
            UIArticles.hasMore =
              UIArticles.currentPage * UIArticles.pageSize <
              UIArticles.allArticlesForInfinite.length;
            UIArticles.renderArticlesPage();
          }
        }
        setTimeout(
          function () {
            this.navigateSearchResult(direction);
          }.bind(this),
          350
        );
        return;
      }
 
      const nodeId = 'article-' + article.id;
      const nodeElement = this.directoryTreeContainer.querySelector(
        '.tree-node[data-node-id="' + nodeId + '"]'
      );
      if (nodeElement) {
        let parent = nodeElement.parentElement;
        while (parent) {
          if (parent.classList && parent.classList.contains('children')) {
            const parentLi = parent.closest('.tree-node.folder');
            if (parentLi) {
              const childrenDiv = parentLi.querySelector(':scope > .children');
              if (childrenDiv) {
                childrenDiv.style.display = 'block';
                const toggleIcon = parentLi.querySelector('.toggle-icon[data-toggle="toggle"]');
                if (toggleIcon) toggleIcon.textContent = '▼';
              }
            }
          }
          parent = parent.parentElement;
        }
 
        const content = nodeElement.querySelector('.tree-node-content');
        if (content) content.classList.add('search-highlight');
 
        nodeElement.scrollIntoView({ behavior: 'smooth', block: 'center' });
        UIHelpers.scrollToElement(targetId);
 
        if (this.searchInput) {
          this.searchInput.placeholder =
            '结果 ' +
            (this.searchCurrentIndex + 1) +
            '/' +
            this.searchResults.length +
            ': ' +
            article.title;
          setTimeout(
            function () {
              if (this.searchInput) this.searchInput.focus();
            }.bind(this),
            50
          );
        }
      }
    }
  },
 
  clearSearchHighlights: function () {
    if (!this.directoryTreeContainer) return;
    const highlights = this.directoryTreeContainer.querySelectorAll(
      '.tree-node-content.search-highlight'
    );
    highlights.forEach(function (el) {
      el.classList.remove('search-highlight');
    });
    if (this.searchInput) {
      const count = this.searchResults.length;
      if (count > 0) {
        this.searchInput.placeholder = '找到 ' + count + ' 个结果';
      } else {
        this.searchInput.placeholder = '🔍 搜索角色...';
      }
    }
  },
 
  expandSearchResults: function () {
    if (this.searchResults.length === 0) return;
 
    const allFolders = this.directoryTreeContainer.querySelectorAll('.tree-node.folder');
    allFolders.forEach(function (folder) {
      const childrenDiv = folder.querySelector(':scope > .children');
      if (childrenDiv) {
        childrenDiv.style.display = 'block';
        const toggleIcon = folder.querySelector('.toggle-icon[data-toggle="toggle"]');
        if (toggleIcon) toggleIcon.textContent = '▼';
      }
    });
 
    if (this.searchCurrentIndex < 0) {
      const firstArticle = this.searchResults[0];
      if (firstArticle && UIArticles) {
        const allVisible = Article.getVisibleArticles();
        const index = allVisible.findIndex(function (a) {
          return a.id === firstArticle.id;
        });
        if (index !== -1) {
          const targetPage = Math.floor(index / UIArticles.pageSize) + 1;
          if (targetPage > UIArticles.currentPage) {
            UIArticles.currentPage = targetPage;
            UIArticles.hasMore =
              UIArticles.currentPage * UIArticles.pageSize <
              UIArticles.allArticlesForInfinite.length;
            UIArticles.renderArticlesPage();
            setTimeout(
              function () {
                this.navigateSearchResult(1);
              }.bind(this),
              350
            );
            return;
          }
        }
      }
      this.navigateSearchResult(1);
    }
  },
};
