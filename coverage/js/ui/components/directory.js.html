// ========== 目录树模块（含可见性控制按钮） ==========
import { Article } from '../../models/article-model.js';
import { AppState } from '../../core/app-state.js';
import { EventBus } from '../../core/event-bus.js';
import { EVENTS } from '../../core/event-constants.js';
import { Utils } from '../../utils.js';
import { UIHelpers } from './helpers.js';
import { UIArticles } from './articles.js';
 
export const UIDirectory = {
  container: null,
  allArticles: [],
 
  init: function (container) {
    console.log('[UIDirectory] 初始化...');
    this.container = container;
    EventBus.on(
      EVENTS.ARTICLE_VISIBILITY_CHANGED,
      function () {
        const articles = Article.getVisibleArticles();
        if (AppState.get('isLoggedIn')) {
          this.updateTree(Article.getAllArticles());
        } else {
          this.updateTree(articles);
        }
      }.bind(this)
    );
    console.log('[UIDirectory] 初始化完成');
  },
 
  /**
   * 保存当前所有文件夹的折叠状态到 localStorage
   */
  _saveFolderStates: function () {
    if (!this.container) return;
    const folderElements = this.container.querySelectorAll('.tree-node.folder');
    folderElements.forEach((folder) => {
      const childrenDiv = folder.querySelector(':scope > .children');
      if (childrenDiv) {
        const isCollapsed = childrenDiv.style.display === 'none';
        const folderName = folder.dataset.name;
        if (folderName) {
          localStorage.setItem('folder-collapsed-' + folderName, JSON.stringify(isCollapsed));
        }
      }
    });
  },
 
  updateTree: function (articles) {
    this._saveFolderStates();
 
    if (!articles || !Array.isArray(articles)) {
      articles = Article.getVisibleArticles();
    }
    const sortedArticles = [...articles].sort((a, b) => a.id - b.id);
    this.allArticles = sortedArticles;
 
    const treeData = Article.buildDirectoryTree(sortedArticles);
    this.container.innerHTML = this.renderTree(treeData);
    this.bindInteractions();
 
    UIArticles.allArticlesForInfinite = sortedArticles;
    console.log('[UIDirectory] 目录树已更新，文章数:', sortedArticles.length);
  },
 
  renderTree: function (nodes, level = 0) {
    if (!nodes || nodes.length === 0) {
      return '<div style="padding: 16px; color: #7a6a58; text-align: center;">空无一人</div>';
    }
    const isAdmin = AppState.get('isLoggedIn');
    let html = '<ul style="list-style: none; padding-left: 0;">';
    for (const node of nodes) {
      const nodeId =
        node.type === 'folder'
          ? 'folder-' + node.name.replace(/[^a-zA-Z0-9\u4e00-\u9fa5]/g, '_')
          : 'article-' + node.articleId;
      const isFolder = node.type === 'folder';
      const hasChildren = isFolder && node.children && node.children.length > 0;
      const storageKey = 'folder-collapsed-' + node.name;
      const stored = localStorage.getItem(storageKey);
      const isCollapsed = stored !== null ? stored === 'true' : true;
      const icon = isFolder ? (isCollapsed ? '📂' : '📁') : '📄';
 
      let visibilityBtn = '';
      let visible = true;
      if (!isFolder) {
        // 直接计算 visible，不使用 articleObj 变量
        const article = Article.allArticles.find((a) => a.id === node.articleId);
        visible = article ? article.visible !== false : true;
        if (isAdmin) {
          visibilityBtn = `<button class="visibility-toggle" data-id="${node.articleId}" data-visible="${visible}" style="background:none;border:none;color:${visible ? '#3a5a2b' : '#5a3e2b'};cursor:pointer;font-size:14px;margin-left:8px;" title="切换可见性">${visible ? '👁️' : '🚫'}</button>`;
        }
      }
 
      html += `<li class="tree-node ${isFolder ? 'folder' : 'article'}" data-node-id="${nodeId}" data-type="${node.type}" data-name="${node.name || ''}" data-article-id="${node.articleId || ''}" data-folder-first-id="${node.firstArticleId || ''}">`;
      html += `<div class="tree-node-content" data-node-id="${nodeId}">`;
 
      if (isFolder && hasChildren) {
        html += `<span class="toggle-icon" data-toggle="toggle">${isCollapsed ? '▶' : '▼'}</span>`;
      } else if (isFolder && !hasChildren) {
        html += `<span class="toggle-icon" style="opacity:0.3;">📭</span>`;
      } else {
        html += `<span class="toggle-icon"></span>`;
      }
 
      html += `<span class="node-icon">${icon}</span>`;
      html += `<span class="node-title">${Utils.escapeHtml(node.name)}</span>`;
 
      if (isAdmin && !isFolder && !visible) {
        html += `<span style="font-size:9px;color:#7a6a58;margin-left:6px;">(访客不可见)</span>`;
      }
 
      if (isAdmin && !isFolder) {
        html += visibilityBtn;
      }
 
      html += '</div>';
 
      if (isFolder && hasChildren && !isCollapsed) {
        html += `<div class="children">${this.renderTree(node.children, level + 1)}</div>`;
      } else if (isFolder && hasChildren && isCollapsed) {
        html += `<div class="children" style="display: none;">${this.renderTree(node.children, level + 1)}</div>`;
      } else if (isFolder && !hasChildren) {
        html += `<div class="children" style="opacity:0.6;"><div style="padding: 8px 16px; font-size:11px; color:#6a5a48;">沉寂的档案架</div></div>`;
      }
 
      html += '</li>';
    }
    html += '</ul>';
    return html;
  },
 
  bindInteractions: function () {
    let clickTimer = null;
    const self = this;
    const container = this.container;
    if (!container) return;
 
    container.removeEventListener('click', this._delegatedClickHandler);
    container.removeEventListener('dblclick', this._delegatedDblClickHandler);
 
    this._delegatedClickHandler = function (e) {
      const toggleBtn = e.target.closest('.visibility-toggle');
      if (toggleBtn) {
        e.stopPropagation();
        const id = parseInt(toggleBtn.dataset.id);
        const currentVisible = toggleBtn.dataset.visible === 'true';
        const newVisible = !currentVisible;
        console.log('[UIDirectory] 切换可见性:', id, newVisible);
        Article.setVisibility(id, newVisible).then(function (success) {
          if (success) {
            toggleBtn.dataset.visible = newVisible;
            toggleBtn.textContent = newVisible ? '👁️' : '🚫';
            toggleBtn.style.color = newVisible ? '#3a5a2b' : '#5a3e2b';
            const parentContent = toggleBtn.closest('.tree-node-content');
            const oldAnnot = parentContent.querySelector('.tree-node-content > span:last-child');
            if (oldAnnot && oldAnnot.textContent === '(访客不可见)') {
              oldAnnot.remove();
            }
            if (!newVisible) {
              const titleSpan = parentContent.querySelector('.node-title');
              const annot = document.createElement('span');
              annot.style.cssText = 'font-size:9px;color:#7a6a58;margin-left:6px;';
              annot.textContent = '(访客不可见)';
              titleSpan.after(annot);
            }
            EventBus.emit(EVENTS.ARTICLE_VISIBILITY_CHANGED, {
              articleId: id,
              visible: newVisible,
              fromRemote: false,
            });
          }
        });
        return;
      }
 
      const toggleIcon = e.target.closest('.toggle-icon[data-toggle="toggle"]');
      if (toggleIcon) {
        e.stopPropagation();
        const nodeLi = toggleIcon.closest('.tree-node.folder');
        if (nodeLi) {
          const childrenDiv = nodeLi.querySelector(':scope > .children');
          if (childrenDiv) {
            const isVisible = childrenDiv.style.display !== 'none';
            childrenDiv.style.display = isVisible ? 'none' : 'block';
            toggleIcon.textContent = isVisible ? '▶' : '▼';
            const folderIcon = nodeLi.querySelector('.node-icon');
            if (folderIcon) folderIcon.textContent = isVisible ? '📁' : '📂';
            const folderName = nodeLi.dataset.name;
            if (folderName) {
              localStorage.setItem('folder-collapsed-' + folderName, JSON.stringify(!isVisible));
            }
          }
        }
        return;
      }
 
      const content = e.target.closest('.tree-node-content');
      if (!content) return;
      const nodeLi = content.closest('.tree-node');
      if (!nodeLi) return;
 
      const nodeData = {
        type: nodeLi.dataset.type,
        articleId: nodeLi.dataset.articleId ? parseInt(nodeLi.dataset.articleId) : null,
        folderFirstId: nodeLi.dataset.folderFirstId ? parseInt(nodeLi.dataset.folderFirstId) : null,
        name: nodeLi.dataset.name,
        nodeId: nodeLi.dataset.nodeId,
      };
 
      if (clickTimer) {
        clearTimeout(clickTimer);
        clickTimer = null;
        self.handleNodeClick(content, nodeData, true);
      } else {
        clickTimer = setTimeout(function () {
          self.handleNodeClick(content, nodeData, false);
          clickTimer = null;
        }, 250);
      }
    };
 
    this._delegatedDblClickHandler = function (e) {
      e.preventDefault();
    };
 
    container.addEventListener('click', this._delegatedClickHandler);
    container.addEventListener('dblclick', this._delegatedDblClickHandler);
    console.log('[UIDirectory] 交互已绑定（含可见性切换）');
  },
 
  handleNodeClick: function (nodeElement, nodeData, isDouble) {
    const type = nodeData.type;
    const articleId = nodeData.articleId;
    const folderFirstId = nodeData.folderFirstId;
    const nodeName = nodeData.name || '';
    const helpers = UIHelpers;
    const articlesModule = UIArticles;
 
    if (!helpers || !articlesModule) return;
 
    if (type === 'article' && articleId) {
      if (isDouble) {
        if (window._UIDetail) {
          window._UIDetail.openDetail(articleId);
        }
      } else {
        const targetId = helpers.generateCardId(articleId);
        const targetEl = document.getElementById(targetId);
        if (targetEl) {
          helpers.scrollToElement(targetId);
          this.setActiveNode(nodeData.nodeId);
        } else {
          const index = articlesModule.allArticlesForInfinite.findIndex(function (a) {
            return a.id === articleId;
          });
          if (index !== -1) {
            const targetPage = Math.floor(index / articlesModule.pageSize) + 1;
            if (targetPage > articlesModule.currentPage) {
              articlesModule.currentPage = targetPage;
              articlesModule.hasMore =
                targetPage * articlesModule.pageSize < articlesModule.allArticlesForInfinite.length;
              articlesModule.renderArticlesPage();
              setTimeout(function () {
                const el = document.getElementById(targetId);
                if (el) helpers.scrollToElement(targetId);
              }, 400);
            }
          }
        }
      }
      return;
    }
 
    if (type === 'folder') {
      if (isDouble) {
        const toggleIcon = nodeElement.querySelector('.toggle-icon[data-toggle="toggle"]');
        if (toggleIcon) toggleIcon.click();
        return;
      }
      if (!folderFirstId) {
        helpers.showNodeWarning('📁 "' + nodeName + '" 文件夹暂无文章');
        return;
      }
      const targetId = helpers.generateCardId(folderFirstId);
      const targetEl = document.getElementById(targetId);
      if (targetEl) {
        helpers.scrollToElement(targetId);
        this.setActiveNode(nodeData.nodeId);
      } else {
        const index = articlesModule.allArticlesForInfinite.findIndex(function (a) {
          return a.id === folderFirstId;
        });
        if (index !== -1) {
          const targetPage = Math.floor(index / articlesModule.pageSize) + 1;
          if (targetPage > articlesModule.currentPage) {
            articlesModule.currentPage = targetPage;
            articlesModule.hasMore =
              targetPage * articlesModule.pageSize < articlesModule.allArticlesForInfinite.length;
            articlesModule.renderArticlesPage();
            setTimeout(function () {
              const el = document.getElementById(targetId);
              if (el) helpers.scrollToElement(targetId);
            }, 400);
          }
        }
      }
    }
  },
 
  setActiveNode: function (nodeId) {
    if (!this.container) return;
    const nodes = this.container.querySelectorAll('.tree-node-content');
    nodes.forEach(function (el) {
      el.classList.remove('active');
    });
    const activeEl = this.container.querySelector(
      '.tree-node-content[data-node-id="' + nodeId + '"]'
    );
    if (activeEl) activeEl.classList.add('active');
  },
};
