// ========== 通知服务 ==========
import { Utils } from '../utils.js';
 
export const NotificationService = {
  // ----- Toast 提示 -----
  showToast(message, isError = false) {
    Utils.showToast(message, isError);
  },
 
  // ----- 确认对话框 -----
  showConfirm(message, callback, cancelCallback) {
    if (confirm(message)) {
      if (typeof callback === 'function') callback();
    } else {
      if (typeof cancelCallback === 'function') cancelCallback();
    }
  },
 
  // ----- 预定义消息库 -----
  messages: {
    visibilityChanged: (visible) => `文章已${visible ? '可见' : '不可见'}`,
    visibilityChangedLocal: (visible) => `文章已${visible ? '可见' : '不可见'}（本地模拟）`,
    visibilityAdminOnly: '需要管理员权限',
    visibilityMadeInvisible: '该文章已被管理员设置为不可见，点击确定关闭当前阅读窗口。',
 
    loginSuccess: '欢迎管理员登录',
    loginFailed: '用户名或密码错误',
    logoutSuccess: '已退出管理员模式',
 
    decoUploadSuccess: (name) => `贴图 "${name}" 上传成功`,
    decoDuplicateSuccess: (name) => `已复制贴图 "${name}"`,
    decoDeleteConfirm: '确定要从贴图库中删除该贴图吗？',
    decoDeleteSuccess: '已删除',
    decoStyleSwitched: (style) => `已切换为${style}样式`,
 
    minimized: '已最小化到左下角',
 
    paletteSaved: '色卡已保存',
    paletteDeleted: '色卡已删除',
    paletteApplied: (name) => `已应用色卡：${name}`,
    paletteDeleteConfirm: '确定删除该色卡吗？',
 
    bgColorApplied: '背景颜色已应用',
    bgColorReset: '背景颜色已重置',
    gradientApplied: '渐变背景已应用',
    solidApplied: '纯色背景已应用',
 
    textureApplied: (size) => `纹理已应用（WebP格式，${size}KB）`,
    textureRemoved: '纹理已移除',
    textureUploadFirst: '请先上传纹理图片',
 
    watermarkApplied: '水印设置已应用',
 
    moduleNotLoaded: '模块未加载',
    operationFailed: '操作失败，请稍后重试',
    operationSuccess: '操作成功',
  },
 
  // ----- 便捷方法 -----
  showVisibilityChanged(visible) {
    this.showToast(this.messages.visibilityChanged(visible));
  },
 
  showMinimized() {
    this.showToast(this.messages.minimized);
  },
 
  showLoginSuccess() {
    this.showToast(this.messages.loginSuccess);
  },
 
  showLoginFailed() {
    this.showToast(this.messages.loginFailed, true);
  },
 
  showLogoutSuccess() {
    this.showToast(this.messages.logoutSuccess);
  },
 
  showDecoUploadSuccess(name) {
    this.showToast(this.messages.decoUploadSuccess(name));
  },
 
  showDecoDuplicateSuccess(name) {
    this.showToast(this.messages.decoDuplicateSuccess(name));
  },
 
  showDecoDeleteSuccess() {
    this.showToast(this.messages.decoDeleteSuccess);
  },
 
  showPaletteSaved() {
    this.showToast(this.messages.paletteSaved);
  },
 
  showPaletteDeleted() {
    this.showToast(this.messages.paletteDeleted);
  },
 
  showPaletteApplied(name) {
    this.showToast(this.messages.paletteApplied(name));
  },
};
